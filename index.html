// Hindi & English Support ke saath Text-to-Speech function speakText(text) { if ('speechSynthesis' in window) { window.speechSynthesis.cancel(); // Pehle se chal rahi awaaz ko rokein const utterance = new SpeechSynthesisUtterance(text); // Hindi Bhasha Detect / Set Karein utterance.lang = 'hi-IN'; // Available Voices me se Hindi voice dhoondhein const voices = window.speechSynthesis.getVoices(); const hindiVoice = voices.find(voice => voice.lang.includes('hi') || voice.lang.includes('HI')); if (hindiVoice) { utterance.voice = hindiVoice; } utterance.rate = 0.9; // Normal bolne ki speed window.speechSynthesis.speak(utterance); } else { alert("Aapke browser me Text-to-Speech support nahi hai."); } } // Browser Voices Load Hone par Setup if ('speechSynthesis' in window) { window.speechSynthesis.onvoiceschanged = () => { window.speechSynthesis.getVoices(); }; }
